# mvesign-web-www Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** mvesign.com opnieuw opbouwen als statische Astro-site in één openbaar repo, die GitHub Actions via FTPS op mijn.host zet.

**Architecture:** Astro bouwt bij elke push kant-en-klare HTML uit markdown (collecties `notes` en `projects`) en een paar `.astro`-templates. De huisstijl komt als npm-pakket uit `mvesign-brand-assets`. Een workflow bouwt, test en uploadt `dist/` met `lftp` naar `public_html`. Op mijn.host draait geen code.

**Tech Stack:** Astro 7.4 (Vite 8, Zod 4, Shiki via Sätteri), Node 26, `@astrojs/rss`, `@astrojs/sitemap`, `@resvg/resvg-wasm`, `node --test`, GitHub Actions, lftp, LiteSpeed (`.htaccess`), Renovate.

**Spec:** `docs/superpowers/specs/2026-11-21-website-design.md`. Lees die eerst; dit plan legt alleen uit *hoe*, de spec zegt *wat* en *waarom*.

```js
// Hulpfuncties voor tests op de gebouwde site in dist/. Draai eerst npm run build.
import { readdir, readFile, access } from 'node:fs/promises';

export const DIST = new URL('../../dist/', import.meta.url);

/** Alle HTML-bestanden in dist/, als paden met / ten opzichte van dist/, gesorteerd. */
export async function htmlFiles() {
  let entries;
  try {
    entries = await readdir(DIST, { recursive: true });
  } catch (error) {
    if (error.code === 'ENOENT') return [];
    throw error;
  }
  return entries
    .map((path) => path.replaceAll('\\', '/'))
    .filter((path) => path.endsWith('.html'))
    .sort();
}

export function readDist(path) {
  return readFile(new URL(path, DIST), 'utf8');
}

export function readDistBytes(path) {
  return readFile(new URL(path, DIST));
}

export async function existsInDist(path) {
  try {
    await access(new URL(path, DIST));
    return true;
  } catch {
    return false;
  }
}
```

```js
const escapeRegExp = (value) => value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

/** Alle waarden van content bij <meta name="key"> of <meta property="key">. */
export function metaAll(html, key) {
  const re = new RegExp(`<meta (?:name|property)="${escapeRegExp(key)}" content="([^"]*)"`, 'g');
  return [...html.matchAll(re)].map((match) => match[1]);
}

/** Alle href's bij <link rel="rel" href="...">. */
export function linkHrefs(html, rel) {
  const re = new RegExp(`<link rel="${escapeRegExp(rel)}" href="([^"]*)"`, 'g');
  return [...html.matchAll(re)].map((match) => match[1]);
}

/** De volledige URL van een pagina in dist/: notes/foo/index.html wordt https://mvesign.com/notes/foo/. */
export function pageUrl(page) {
  return `https://mvesign.com/${page.replace(/index\.html$/, '')}`;
}
```

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import { htmlFiles, readDist, existsInDist, metaAll, linkHrefs, pageUrl } from './helpers/dist.mjs';

const pages = await htmlFiles();

for (const page of pages) {
  test(`${page}: <head> voor zoekmachines en delen`, async () => {
    const html = await readDist(page);
    assert.equal(html.match(/<title>/g)?.length, 1, 'precies één <title>');
    assert.equal(metaAll(html, 'description').length, 1, 'description');
    for (const key of ['og:title', 'og:description', 'og:type', 'og:site_name']) {
      assert.equal(metaAll(html, key).length, 1, key);
    }
    assert.deepEqual(metaAll(html, 'og:locale'), ['nl_NL']);
    assert.deepEqual(metaAll(html, 'og:image'), ['https://mvesign.com/og-default.png']);
    if (page === '404.html') {
      assert.deepEqual(metaAll(html, 'robots'), ['noindex']);
      assert.deepEqual(linkHrefs(html, 'canonical'), []);
      assert.deepEqual(metaAll(html, 'og:url'), []);
    } else {
      assert.deepEqual(linkHrefs(html, 'canonical'), [pageUrl(page)]);
      assert.deepEqual(metaAll(html, 'og:url'), [pageUrl(page)]);
```

```ts
// De deelafbeelding (1200 × 630): het vlak van de balk, de rand eronder en het woordmerk in het
// midden. Kleuren en woordmerk komen uit de huisstijl; het woordmerk bestaat uit paden, dus
// resvg heeft geen lettertype nodig.
import { readFile } from 'node:fs/promises';
import { pathToFileURL } from 'node:url';
import { initWasm, Resvg } from '@resvg/resvg-wasm';
import tokens from 'mvesign-brand-assets/tokens.json';
import wordmark from 'mvesign-brand-assets/logo/wordmark.svg?raw';

const WIDTH = 1200;
const HEIGHT = 630;
const EDGE = 24;
const WORDMARK_WIDTH = 600;

export function ogSvg(): string {
  const { bar, bar_edge } = tokens.color.light;
  const viewBox = /viewBox="([^"]+)"/.exec(wordmark);
  if (!viewBox) throw new Error('wordmark.svg heeft geen viewBox');
  const [, , vbWidth, vbHeight] = viewBox[1].trim().split(/\s+/).map(Number);
  const scale = WORDMARK_WIDTH / vbWidth;
  const x = (WIDTH - WORDMARK_WIDTH) / 2;
  const y = (HEIGHT - EDGE - vbHeight * scale) / 2;
  const inner = wordmark.replace(/^[\s\S]*?<svg[^>]*>/, '').replace(/<\/svg>\s*$/, '');
  return [
    `<svg xmlns="http://www.w3.org/2000/svg" width="${WIDTH}" height="${HEIGHT}" viewBox="0 0 ${WIDTH} ${HEIGHT}">`,
    `<rect width="${WIDTH}" height="${HEIGHT}" fill="${bar}"/>`,
    `<rect y="${HEIGHT - EDGE}" width="${WIDTH}" height="${EDGE}" fill="${bar_edge}"/>`,
    `<g transform="translate(${x} ${y}) scale(${scale})">${inner}</g>`,
    '</svg>',
  ].join('');
}

let ready: Promise<void> | undefined;

export async function renderOgPng(): Promise<Uint8Array> {
  // De build draait vanuit de map van het project (npm run build), dus daar staat node_modules.
  const wasmUrl = new URL('node_modules/@resvg/resvg-wasm/index_bg.wasm', pathToFileURL(`${process.cwd()}/`));
  ready ??= readFile(wasmUrl).then((wasm) => initWasm(wasm));
  await ready;
  return new Resvg(ogSvg(), { fitTo: { mode: 'width', value: WIDTH } }).render().asPng();
}
```

```ts
export interface Dated {
  id: string;
  data: { date: Date };
}

export interface Yearly {
  data: { year: number; title: string };
}

/** Nieuwste eerst; bij dezelfde datum op id, zodat de volgorde vast is. */
export function byDateDesc(a: Dated, b: Dated): number {
  return b.data.date.getTime() - a.data.date.getTime() || a.id.localeCompare(b.id);
}

/** Nieuwste jaar eerst, daarna op titel. */
export function byYearDescThenTitle(a: Yearly, b: Yearly): number {
  return b.data.year - a.data.year || a.data.title.localeCompare(b.data.title, 'nl', { sensitivity: 'base' });
}
```

```yaml
          set ssl:verify-certificate true
          set net:max-retries 2
```
